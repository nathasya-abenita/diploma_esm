\section{Introduction}

\begin{frame}
    \begin{block}{Motivation}
        \begin{itemize}
            \item Intensification of the hydrological cycle is a known effect of a warming climate.%, including rainfall extremes increase.
            \item Uncertainty remains in understanding changes to \textbf{short-duration} and \textbf{convective} events.
            \item Short-duration rainfall extremes are responsible for fatalities through flash floods and landslides.
        \end{itemize}
        
    \end{block}

    \begin{block}{Goals}
        \begin{itemize}
            \item Review observational and modelling evidences for the intensification of these extremes
            \item Understand future floods risks
        \end{itemize}
    \end{block}
\end{frame}

\section{Thermodynamics Baseline}

\subsection{Clausius-Clapeyron (CC) Relation}
\begin{frame}
    \begin{itemize}
        \item Saturation specific humidity of the atmosphere increases at a rate of $\sim7\%$ per degree warming near the Earth's surface. 
        \item Assuming constant relative humidity, specific humidity of air also increases at roughly the same rate.
        \item As rainfall extremes are limited by the amount of atmospheric moisture available, changes to rainfall intensities are expected to scale with the CC relation.
    \end{itemize}
\end{frame}

\subsection{Apparent and Climate Scaling}
\begin{frame}
    \begin{definition}
        \begin{itemize}
            \item \textbf{Apparent scaling:} Between extreme rainfall intensities and temperature from \textbf{observed} short-term climate variability.
            \item \textbf{Climate scaling:} How extreme rainfall will respond in a changing climate.
        \end{itemize}
    \end{definition}
\end{frame}

\subsection{Findings from Convective Permitting Models (CPMs)}
\begin{frame}
    \begin{itemize}
        \item CPMs use very-high-resolution grids, allowing them to explicitly resolve cloud dynamical processes.
        \item CPMs consistently project higher intensification of sub-daily extremes (at or above CC) compared to coarser models. \pause
        \item There is increase in peak intensities and frequencies for very rare rainfall extremes, decrease in frequency for moderate and light rainfall hours.
        \begin{itemize}
            \item Global precipitation is constrained by the global energy budget!
        \end{itemize}
    \end{itemize}
\end{frame}

\section{Conclusion}
\begin{frame}
    \begin{itemize}
        \item Short-duration extremes are intensifying at CC or super-CC rates due to \textbf{thermodynamic and dynamical feedbacks}.
        \item There is an urgent need for adaptation measures, especially in urban areas, to manage the \textbf{non-linear increase} in flood risk.
    \end{itemize}
\end{frame}
